```lean
theorem FSC.Peano2.hasDerivWithinAt_noiseAverage
    {V : Type*} [NormedAddCommGroup V] [NormedSpace ℝ V]
    [MeasurableSpace V] [BorelSpace V] [CompleteSpace V] [SecondCountableTopology V]
    {f : V → ℝ} {x : V} {l : V →L[ℝ] ℝ} {B : V →L[ℝ] V →L[ℝ] ℝ}
    (hp : FSC.Peano2 f x l B) (hf : Measurable f)
    (hb : ∃ M : ℝ, ∀ z, ‖f z‖ ≤ M)
    (μ : MeasureTheory.Measure V) [MeasureTheory.IsProbabilityMeasure μ]
    (hmean : ∫ y, y ∂μ = 0)
    (hsecond : MeasureTheory.Integrable (fun y : V ↦ ‖y‖ ^ 2) μ)
    {v : ℝ → V} {v₀ : V} (hvzero : v 0 = 0)
    (hv : Filter.Tendsto (fun s : ℝ ↦ s⁻¹ • v s) (𝓝[>] (0 : ℝ)) (𝓝 v₀)) :
    HasDerivWithinAt (fun s : ℝ ↦ ∫ y, f (x + v s + Real.sqrt s • y) ∂μ)
      (l v₀ + (1 / 2 : ℝ) * ∫ y, B y y ∂μ) (Set.Ici 0) 0
```
